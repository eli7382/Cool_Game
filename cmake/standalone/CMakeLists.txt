cmake_minimum_required(VERSION 3.31)
project(BreakwaterStandalone LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT RAYLIB_SOURCE_DIR)
  message(FATAL_ERROR "Set RAYLIB_SOURCE_DIR to the official raylib 5.5 source checkout")
endif()
get_filename_component(GAME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${GAME_ROOT}/cmake/StageRaylibWorldBatches.cmake")
set(STAGED_RAYLIB_SOURCE "${CMAKE_BINARY_DIR}/raylib-world-batches")
stage_raylib_world_batches("${RAYLIB_SOURCE_DIR}" "${STAGED_RAYLIB_SOURCE}")
add_subdirectory("${STAGED_RAYLIB_SOURCE}" raylib)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${GAME_ROOT}/src/breakwater/*.cpp")
add_executable(game ${GAME_SOURCES})
target_link_libraries(game PRIVATE raylib)
target_compile_definitions(game PRIVATE _CRT_SECURE_NO_WARNINGS BW_WORLD_SPACE_BATCHES_FIXED=1)
set(BW_PORSCHE_ASSET_HEADER "" CACHE FILEPATH "Optional asset metadata header for isolated candidate validation")
if(BW_PORSCHE_ASSET_HEADER)
  target_compile_definitions(game PRIVATE "BW_PORSCHE_ASSET_HEADER=\"${BW_PORSCHE_ASSET_HEADER}\"")
endif()
if(MSVC)
  target_compile_options(game PRIVATE /W4 /EHsc)
else()
  target_compile_options(game PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Wno-misleading-indentation)
endif()
set_target_properties(game PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
